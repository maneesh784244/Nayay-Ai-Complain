function getReply(t){
  const l=t.toLowerCase()
  if(l.includes('fraud')||l.includes('online')||l.includes('paise'))return aiR.fraud[Math.floor(Math.random()*aiR.fraud.length)]
  if(l.includes('consumer')||l.includes('amazon')||l.includes('refund'))return aiR.consumer[Math.floor(Math.random()*aiR.consumer.length)]

  return aiR.default[turn++%aiR.default.length]
}
